defmodule Steward do
  use GenServer

  # client

  def start_link() do
    GenServer.start_link(__MODULE__, [])
  end

  # generser.start_link starts with supervision tree, start just goes

  def get_log(pid) do
    GenServer.call(pid, :log)
  end

  defp get_end(pid) do
    receive do
      {:DOWN, ospid, :process, _pid, :normal} ->
        [output: "process exited normally", ospid: ospid, timestamp: current_time()]
    after
      500 -> nil
    end
  end

  def get_log_stream(pid) do
    Stream.resource(
      fn -> get_log(pid) end,
      fn log ->
        end_check = get_end(pid)

        log =
          if end_check != nil do
            {log ++ [end_check], :done}
          else
            log
          end

        case log do
          [first | _] when length(log) == 1 ->
            get_new =
              Task.async(fn ->
                get_new_elements(pid, first)
              end)

            {[first], get_new}

          [first | rest] ->
            {[first], rest}

          %Task{} ->
            case Task.yield(log) do
              {:ok, {nil, cycle}} ->
                get_new = Task.async(fn -> get_new_elements(pid, cycle) end)
                {[], get_new}

              {:ok, [resp_first | resp_rest]} ->
                {[resp_first], resp_rest}
            end

          {final_out, :done} ->
            {final_out, :done}

          :done ->
            {:halt, []}
        end
      end,
      fn _ -> :ok end
    )
  end

  def get_new_elements(pid, latest) do
    response = GenServer.call(pid, {:next_entries, latest})

    if response == nil do
      Process.sleep(500)
      {nil, latest}
    else
      response
    end
  end

  def kill(pid) do
    GenServer.stop(pid)
  end

  def set_dir(opts, dir) do
    absdir = Path.expand(dir)

    if not File.exists?(absdir) do
      raise(ArgumentError, "directory does not exist")
    end

    [{:cd, absdir} | opts]
  end

  def set_sync(opts, sync \\ false) do
    if sync do
      [:sync | opts]
    else
      opts
    end
  end

  def set_env(opts, :clear) do
    [{:env, [:clear]} | opts]
  end

  def set_env(opts, env) when is_list(env) do
    [{:env, env} | opts]
  end

  ### check to be sure that set env isnt called more than once?
  def set_env(opts, env) when is_binary(env) or is_tuple(env) do
    [{:env, [env]} | opts]
  end

  def run_command(pid, cmd, opts \\ []) when is_binary(cmd) do
    capture_fun = fn std, ospid, d -> make_log_entry(std, ospid, d, pid) end
    capture_output = [:monitor, {:stdout, capture_fun}, {:stderr, capture_fun}]

    opts = capture_output ++ opts

    :exec.run(cmd, opts)
  end

  # server

  @impl true
  def init(log) do
    first_entry = [[output: "Steward initialised", timestamp: current_time()] | log]
    {:ok, first_entry}
  end

  @impl true
  def handle_cast({:add_element, element}, log) do
    new_log = [element | log]
    {:noreply, new_log}
  end

  @impl true
  def handle_call({:next_entries, latest_entry}, _from, log) do
    {:ok, latest_timestamp} =
      latest_entry[:timestamp]
      |> Time.from_iso8601()

    new_entries =
      log
      |> Enum.filter(fn x ->
        {:ok, x_timestamp} =
          x[:timestamp]
          |> Time.from_iso8601()

        case Time.compare(latest_timestamp, x_timestamp) do
          :lt -> true
          _ -> false
        end
      end)

    case new_entries do
      [] -> {:reply, nil, log}
      _ -> {:reply, new_entries, log}
    end
  end

  @impl true
  def handle_call(:log, _from, log) do
    output =
      log
      |> Enum.reverse()

    {:reply, output, log}
  end

  def handle_info({:message}, log) do
    new_log = log ++ message
    {:noreply, new_log}
  end

  # helper functions

  defp make_log_entry(std, ospid, data, pid) do
    entry = [output_type: std, output: data, ospid: ospid, timestamp: current_time()]
    GenServer.cast(pid, {:add_element, entry})
  end

  defp current_time() do
    Time.utc_now() |> Time.to_iso8601()
  end
end
